FUNCTION_BLOCK FB_SafeMotionLimiter
VAR_INPUT
    InPos   : LREAL;        // 目標輸入位置
    MaxVel  : LREAL := 50.0;   // 最大速度 (deg/s)
    MaxAcc  : LREAL := 1000.0;   // 最大加速度 (deg/s²)
    dt1      : LREAL := 0.001;    // Task 週期時間 (秒)，預設為 1ms
    Kp      : LREAL := 50.0;      // 目標追隨比例增益
END_VAR

VAR_OUTPUT
    OutPos : LREAL;        // 平滑輸出位置
END_VAR

VAR
    PrevPos : LREAL := 0.0; // 上一迭代位置
    PrevVel : LREAL := 0.0; // 上一迭代速度
    TargetVel : LREAL;
    TargetAcc : LREAL;
    VelDelta : LREAL;
    Error : LREAL;
END_VAR

// 計算目標誤差
Error := InPos - PrevPos;
TargetVel := Error * Kp;

// 限制速度
IF TargetVel > MaxVel THEN
    TargetVel := MaxVel;
ELSIF TargetVel < -MaxVel THEN
    TargetVel := -MaxVel;
END_IF

// 計算加速度
VelDelta := TargetVel - PrevVel;
IF dt1<>0 THEN
	TargetAcc := VelDelta / dt1;
END_IF

// 限制加速度
IF TargetAcc > MaxAcc THEN
    TargetAcc := MaxAcc;
END_IF

IF TargetAcc < -MaxAcc THEN
    TargetAcc := -MaxAcc;
END_IF

// 更新速度與位置
PrevVel := PrevVel + TargetAcc * dt1;
OutPos := PrevPos + PrevVel * dt1;

// 儲存位置供下迭代使用
PrevPos := OutPos;
